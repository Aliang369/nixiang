### 1.1 层级常量与 safeLayer

```js
// L389

// L390-401
function safeLayer(z) {
    var max = 32760;
    try { var m = Number(cc.macro && cc.macro.MAX_ZINDEX); if (isFinite(m) && m > max) max = Math.min(m, 999999999); } catch (e) {}
    if (z === UI_LAYER.auth)  return max;        // 卡密遮罩：永远最顶

```

**每个字段值**：`ball=32000`、`panel=32010`、`auth=32760`。注意 `safeLayer` **不是直接返回 32000/32010/32760**，而是把它们当"身份标签"映射到 `cc.macro.MAX_ZINDEX` 附近：`auth → max`、`panel → max-2`、`ball → max-4`（max 默认 32760，若引擎常量更大则取引擎值，上限 999999999）。游戏自身也用 `cc.macro.MAX_ZINDEX` 放顶层 HUD（L387-388 注释），补丁沿用同一常量 + `siblingIndex` 置顶，避免 OxygenOS/ColorOS 多 Canvas 下"节点在但被全屏 UI 盖住"。

## 7. 防退出 guardExit（L2285-2301，L2279-2284 是它的边界注释）

```js
function guardExit() {
    try {
        var g = window.gdkjsb;
        if (!g || g.__wxGuard || typeof g.exitProgram !== 'function') return;    // 无桥/已装/不是函数 → 不装
        var orig = g.exitProgram;
        g.exitProgram = function () {
            if (CFG.noExit === false) return orig.apply(this, arguments);        // 用户关了防退出 → 走原函数
            try {
                window.__WX_EXIT_BLOCKED__ = (window.__WX_EXIT_BLOCKED__ || 0) + 1;   // 拦截计数
                window.__WX_EXIT_STACK__ = (new Error()).stack || '';                 // 拦一次记一次栈
            } catch (e) {}
            return undefined;                                                      // 吞掉返回值
        };
        g.__wxGuard = true;              // 重入保护标记，打在桥对象上
        window.__WX_NOEXIT__ = true;     // 对外状态位
```

**具体拦了什么**：只拦 `window.gdkjsb.exitProgram` 这**一个** JS bridge 方法，即"通过 JS bridge 发起的退出"。拦下时返回 `undefined`（不执行原函数、不抛错），并把计数写 `__WX_EXIT_BLOCKED__`、调用栈写 `__WX_EXIT_STACK__`。开关字段 `CFG.noExit` 默认 **true**：L2312 `noExit: storageGet('wx_noExit', '1') !== '0'`（即只有显式存 `'0'` 才关）。`g.__wxGuard = true` 防重复包装。
