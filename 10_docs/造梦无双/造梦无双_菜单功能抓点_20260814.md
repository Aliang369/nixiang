### 3.1 四段 hook 正文（BSS `@≈+0x25d00`）

```javascript
// get_ret(1) 无限技能 — 绕过释放校验
if (this.mPlayer.isDead() || this.mPlayer.mBuffMgr && !this.mPlayer.mBuffMgr.isCanReleaseSkill()) return "1";
var isChecked_RET = jsb.reflection.callStaticMethod("com/xm4399/zmws/AppActivity", "get_ret", "(I)Z", 1);
if (isChecked_RET) { return null; }

// get_ret(2) 法宝技能 — 能量扣减旁路
var e = this.mPlayer.mProp.getMagicCurEneryCount() - 1;
var isChecked_RET = jsb.reflection.callStaticMethod("com/xm4399/zmws/AppActivity", "get_ret", "(I)Z", 2);
if (isChecked_RET) { return null; }

// get_ret(3) 超级秒杀
getAtk = function(e, i) {
  var isChecked_RET = jsb.reflection.callStaticMethod("com/xm4399/zmws/AppActivity", "get_ret", "(I)Z", 3);
  if (isChecked_RET) { return 9999999999999; }
  // ...
}

// get_ret(4) 超级无敌
getAtk = function(e, t) {
  var isChecked_RET = jsb.reflection.callStaticMethod("com/xm4399/zmws/AppActivity", "get_ret", "(I)Z", 4);
  if (isChecked_RET) { return 0; }
  // ...
}
```

另见壳侧把 `copyHandler.addCheat` 改成空 `return;`（反举报相关，星光可选对齐）。

### 3.2 机制链

1. WebView `Changes` → 写 `{pkg}_preferences.xml` `"0"`–`"3"`。  
2. 同步 BSS 旗标 `@+0x2798`。  
3. `libSDKloading` **Dobby** 钩 Cocos **`evalString`/`jsb`**，按 `get_ret` 注入上表 JS。  
4. **`libcocos2djs` 映像本身不改玩法字节** → 星光应走 **CocosJsBridge / eval 同构**，不是 Keypatch。
