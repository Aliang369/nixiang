235:     void* ra = __builtin_return_address(0);
236:     if (g_ready && is_target(path) && ra_in_fg(ra)) {
237:         if (g_spoofOn) {
238:             /* 伪装：fd 指向 /data/data/<pkg>/files/fg_orig.apk（完整原厂 base.apk） */
239:             int sfd = o_open(kSpoofPath, flags & ~(O_CREAT | O_EXCL | O_TRUNC));
240:             if (sfd >= 0) {
241:                 g_spoofFd = sfd;
242:                 int n = ++g_hits;
243:                 if (n <= 12) logger("FG open(SPOOF) #%d %s -> fd=%d ra=%p", n, path, sfd, ra);
244:                 return sfd;
245:             }
246:             logger("FG open(SPOOF) 失败 errno=%d ⇒ 回退真实 base.apk", errno);
247:         }
248:         int n = ++g_hits;
249:         if (n <= 12) logger("FG open(real) #%d %s ra=%p", n, path, ra);

251:     if (flags & (O_CREAT | O_TMPFILE)) {
252:         va_list ap; va_start(ap, flags); int m = va_arg(ap, int); va_end(ap);
253:         return o_open(path, flags, m);
254:     }
255:     return o_open(path, flags);
256: }
